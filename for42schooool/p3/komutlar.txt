# --- K3D VE CLUSTER KURULUMU ---
# 1. k3d'yi kur (Eğer kurulu değilse)
brew install k3d

# 2. Port yönlendirmeleriyle birlikte K3d cluster'ını ayağa kaldır
k3d cluster create iot-cluster --port "8888:80@loadbalancer" --port "8080:8080@loadbalancer"

# --- ARGOCD KURULUMU ---
# 3. Gerekli Namespace'leri (İsim alanlarını) oluştur
kubectl create namespace argocd
kubectl create namespace dev

# 4. ArgoCD'yi Kubernetes üzerine kur
kubectl apply -n argocd -f https://raw.githubusercontent.com/argoproj/argo-cd/stable/manifests/install.yaml

# 5. ArgoCD'nin ilk admin şifresini çek
kubectl -n argocd get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d; echo

# 6. Arayüze (UI) erişmek için port yönlendirmesi yap (Terminali açık bırak)
kubectl port-forward svc/argocd-server -n argocd 8080:443

# (Tarayıcıdan https://localhost:8080 adresine gidilir, admin ve kopyalanan şifre ile girilir.)
